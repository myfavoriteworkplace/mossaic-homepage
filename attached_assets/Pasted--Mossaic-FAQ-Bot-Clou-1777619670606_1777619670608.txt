/** Returns true if the request origin is allowed. */
function isAllowedOrigin(request) {
  const origin  = request.headers.get("Origin")  || "";
  const referer = request.headers.get("Referer") || "";

  // During Wrangler local dev there may be no origin header — allow it.
  if (!origin && !referer) return true;

  return ALLOWED_ORIGINS.some(
    (o) => origin.startsWith(o) || referer.startsWith(o)
  );
}

/** Returns true if the message contains a hard off-topic pattern. */
function isOffTopic(text) {
  const lower = text.toLowerCase();
  return OFF_TOPIC_PATTERNS.some((re) => re.test(lower));
}

/** Returns true if the message looks like a prompt-injection attempt. */
function isJailbreak(text) {
  return JAILBREAK_SOFT.some((re) => re.test(text));
}

/** Sanitise user message — strip excessive whitespace, trim. */
function sanitise(text) {
  return text.replace(/\s+/g, " ").trim();
}

/**
 * Build the messages array for Groq.
 * history  — array of {role, content} from the frontend (last N turns)
 * userMsg  — the current user message (already sanitised)
 */
function buildMessages(history, userMsg) {
  const systemMsg = { role: "system", content: SYSTEM_PROMPT };

  // Keep only the last MAX_HISTORY_TURNS turns and ensure roles alternate
  // correctly (Groq is strict about this).
  const safeHistory = (Array.isArray(history) ? history : [])
    .slice(-MAX_HISTORY_TURNS * 2) // each turn = 2 messages
    .filter(
      (m) =>
        m &&
        typeof m.role === "string" &&
        typeof m.content === "string" &&
        ["user", "assistant"].includes(m.role)
    )
    .map((m) => ({ role: m.role, content: sanitise(m.content).slice(0, 600) }));

  return [systemMsg, ...safeHistory, { role: "user", content: userMsg }];
}

/** Post to Groq and return the assistant's reply string. */
async function callGroq(messages, apiKey) {
  const res = await fetch(GROQ_ENDPOINT, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${apiKey}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      model:       GROQ_MODEL,
      messages,
      temperature: TEMPERATURE,
      max_tokens:  MAX_TOKENS,
      top_p:       0.9,
      // Penalise repetition — prevents the model looping the same phrase
      frequency_penalty: 0.3,
      presence_penalty:  0.1,
    }),
  });

  if (!res.ok) {
    const errText = await res.text();
    console.error("Groq error:", res.status, errText);
    throw new Error(`Groq ${res.status}`);
  }

  const data = await res.json();
  const reply = data?.choices?.[0]?.message?.content;

  if (!reply || typeof reply !== "string" || reply.trim().length === 0) {
    throw new Error("Empty Groq response");
  }

  return reply.trim();
}

/** Return a JSON response with CORS headers. */
function jsonResponse(body, status = 200) {
  return new Response(JSON.stringify(body), {
    status,
    headers: { "Content-Type": "application/json", ...CORS },
  });
}

/* ══════════════════════════════════════════════════════════════════════
   MAIN HANDLER
══════════════════════════════════════════════════════════════════════ */
export default {
  async fetch(request, env) {
    /* ── Layer 0: CORS preflight ──────────────────────────────────────── */
    if (request.method === "OPTIONS") {
      return new Response(null, { status: 204, headers: CORS });
    }

    /* ── Layer 0: Health check ────────────────────────────────────────── */
    if (request.method === "GET") {
      return jsonResponse({
        status: "ok",
        bot: "Mossie — Mossaic FAQ assistant",
        model: GROQ_MODEL,
        version: "2.0.0",
      });
    }

    /* ── Only POST beyond this point ─────────────────────────────────── */
    if (request.method !== "POST") {
      return jsonResponse({ error: "Method not allowed" }, 405);
    }

    /* ── Layer 1: Origin / Referer guard ─────────────────────────────── */
    // Comment out the block below during local testing if needed.
    if (!isAllowedOrigin(request)) {
      return jsonResponse({ error: "Forbidden" }, 403);
    }

    /* ── Parse body ───────────────────────────────────────────────────── */
    let body;
    try {
      body = await request.json();
    } catch {
      return jsonResponse({ error: "Invalid JSON body" }, 400);
    }

    const rawMessage = body.message ?? "";
    const history    = body.history  ?? []; // [{role, content}, ...]

    /* ── Layer 2: Input sanity ────────────────────────────────────────── */
    if (!rawMessage || typeof rawMessage !== "string") {
      return jsonResponse({ reply: MSG_EMPTY });
    }

    const message = sanitise(rawMessage);

    if (message.length === 0) {
      return jsonResponse({ reply: MSG_EMPTY });
    }

    if (message.length > MAX_INPUT_CHARS) {
      return jsonResponse({ reply: MSG_TOO_LONG });
    }

    /* ── Layer 3: Hard topic guard ────────────────────────────────────── */
    if (isOffTopic(message)) {
      return jsonResponse({ reply: MSG_BOUNDARY });
    }

    /* ── Layer 4: Jailbreak guard ─────────────────────────────────────── */
    if (isJailbreak(message)) {
      return jsonResponse({ reply: MSG_JAILBREAK });
    }

    /* ── Layer 5: Groq API call ───────────────────────────────────────── */
    try {
      const messages = buildMessages(history, message);
      const reply    = await callGroq(messages, env.GROQ_API_KEY);

      /* Post-processing:
         - Strip any accidental markdown headers the model adds
         - Ensure the reply doesn't leak system prompt content
         - Trim to a safe length if the model ignored max_tokens   */
      const cleanReply = reply
        .replace(/^#{1,3}\s+/gm, "")          // strip ## headers
        .replace(/\*\*(.*?)\*\*/g, "$1")       // strip **bold**
        .replace(/^\s*[-*]\s/gm, "• ")         // normalise bullets
        .slice(0, 800)                          // hard length cap
        .trim();

      return jsonResponse({ reply: cleanReply });

    } catch (err) {
      console.error("Worker error:", err);
      return jsonResponse({ reply: MSG_ERROR });
    }
  },
};

/* ── B1. Context injector ────────────────────────────────────────────
   Builds a small context string injected into the user message before
   it's sent to the Worker. The Worker's system prompt references
   "the system will inject the current time and date" — this is how.
   The model sees it as part of the user turn, reads naturally.
────────────────────────────────────────────────────────────────────── */

const DAYS    = ["Sunday","Monday","Tuesday","Wednesday","Thursday","Friday","Saturday"];
const MONTHS  = ["January","February","March","April","May","June","July","August","September","October","November","December"];

/**
 * Returns a small context prefix to prepend to the user message.
 * Example output: "[Context: Tuesday, 29 April 2026, 10:34 AM IST] "
 * The model uses this for time/date/day-of-week awareness.
 */
function buildContextPrefix() {
  const now  = new Date();
  const day  = DAYS[now.getDay()];
  const date = now.getDate();
  const mon  = MONTHS[now.getMonth()];
  const yr   = now.getFullYear();
  const h    = now.getHours();
  const m    = String(now.getMinutes()).padStart(2, "0");
  const ampm = h >= 12 ? "PM" : "AM";
  const h12  = h % 12 || 12;
  const isWeekend = now.getDay() === 0 || now.getDay() === 6;

  return (
    `[Context: ${day}, ${date} ${mon} ${yr}, ` +
    `${h12}:${m} ${ampm} IST, ` +
    `${isWeekend ? "weekend" : "weekday"}] `
  );
}

/**
 * Wrap the user message with the context prefix.
 * Call this just before sending to the Worker.
 *
 * @param {string} userMessage — raw user input
 * @returns {string}           — prefixed message for the Worker
 */
export function injectContext(userMessage) {
  return buildContextPrefix() + userMessage.trim();
}


/* ── B2. Name capture ────────────────────────────────────────────────
   Detects if the user introduces their name and stores it.
   The stored name is injected into the context prefix so Mossie
   can use it naturally.
────────────────────────────────────────────────────────────────────── */

let _userName = null;

const NAME_PATTERNS = [
  /\bmy name is\s+([A-Za-z]+)/i,
  /\bi(?:'m| am)\s+(?:dr\.?\s+)?([A-Za-z]+)/i,
  /\bcall me\s+([A-Za-z]+)/i,
  /\bthis is\s+(?:dr\.?\s+)?([A-Za-z]+)/i,
];

/**
 * Attempt to extract a name from the user's message.
 * Stores it in _userName if found. Safe to call on every message.
 *
 * @param {string} message
 */
export function captureUserName(message) {
  if (_userName) return; // already captured — don't overwrite
  for (const pattern of NAME_PATTERNS) {
    const match = message.match(pattern);
    if (match && match[1] && match[1].length > 1 && match[1].length < 30) {
      _userName = match[1].charAt(0).toUpperCase() + match[1].slice(1).toLowerCase();
      return;
    }
  }
}

/** Returns the captured name or null. */
export function getUserName() { return _userName; }

/** Resets the name (call on chat reset). */
export function clearUserName() { _userName = null; }

/**
 * Extends buildContextPrefix to include the user name if captured.
 * Replace your injectContext call with this version once name capture
 * is wired in.
 *
 * @param {string} userMessage
 * @returns {string}
 */
export function injectContextWithName(userMessage) {
  const nameClause = _userName ? `, user's name is ${_userName}` : "";
  const now  = new Date();
  const day  = DAYS[now.getDay()];
  const date = now.getDate();
  const mon  = MONTHS[now.getMonth()];
  const yr   = now.getFullYear();
  const h    = now.getHours();
  const m    = String(now.getMinutes()).padStart(2, "0");
  const ampm = h >= 12 ? "PM" : "AM";
  const h12  = h % 12 || 12;
  const isWeekend = now.getDay() === 0 || now.getDay() === 6;

  return (
    `[Context: ${day}, ${date} ${mon} ${yr}, ` +
    `${h12}:${m} ${ampm} IST, ` +
    `${isWeekend ? "weekend" : "weekday"}` +
    `${nameClause}] ` +
    userMessage.trim()
  );
}


/* ── B3. Frustration detector ────────────────────────────────────────
   Detects frustrated or confused messages and prepends an empathy
   note to the context so the model leads with empathy automatically.
────────────────────────────────────────────────────────────────────── */

const FRUSTRATION_PATTERNS = [
  /\b(don'?t understand|can'?t figure|confused|confusing|doesn'?t make sense|makes no sense)\b/i,
  /\b(tried everything|nothing works|not working|broken|bug|error)\b/i,
  /\b(frustrated|annoying|annoyed|useless|waste of time|pointless)\b/i,
  /\b(again and again|keep trying|still not|why (isn'?t|won'?t|can'?t))\b/i,
  /!{2,}|\?{2,}/,  // multiple ! or ? signals frustration
];

/**
 * Returns true if the message sounds frustrated or confused.
 * @param {string} message
 */
export function isFrustrated(message) {
  return FRUSTRATION_PATTERNS.some((re) => re.test(message));
}

/**
 * If frustrated, prepend an empathy instruction to the context
 * so the model always leads with empathy before the answer.
 *
 * @param {string} contextMessage — already context-injected message
 * @param {string} rawMessage     — original user message
 * @returns {string}
 */
export function injectEmpathyIfNeeded(contextMessage, rawMessage) {
  if (isFrustrated(rawMessage)) {
    return "[Note: User sounds frustrated or confused. Lead with empathy before answering.] " + contextMessage;
  }
  return contextMessage;
}


/* ── B4. Language detector ───────────────────────────────────────────
   Detects if the user is writing in Malayalam or Hindi.
   Injects a language instruction into the context so the model
   responds in the correct language for the greeting.
────────────────────────────────────────────────────────────────────── */

// Malayalam Unicode range: 0D00–0D7F
const MALAYALAM_RE = /[\u0D00-\u0D7F]/;
// Hindi/Devanagari Unicode range: 0900–097F
const HINDI_RE     = /[\u0900-\u097F]/;

/**
 * Detects the script of the user message.
 * @param {string} message
 * @returns {"malayalam" | "hindi" | "english"}
 */
export function detectLanguage(message) {
  if (MALAYALAM_RE.test(message)) return "malayalam";
  if (HINDI_RE.test(message))     return "hindi";
  return "english";
}

/**
 * If a non-English script is detected, prepend a language instruction.
 * @param {string} contextMessage
 * @param {string} rawMessage
 * @returns {string}
 */
export function injectLanguageIfNeeded(contextMessage, rawMessage) {
  const lang = detectLanguage(rawMessage);
  if (lang === "malayalam") {
    return "[Note: User is writing in Malayalam. Greet in Malayalam. Ask if they want to continue in English or Malayalam.] " + contextMessage;
  }
  if (lang === "hindi") {
    return "[Note: User is writing in Hindi. Greet in Hindi. Ask if they want to continue in English or Hindi.] " + contextMessage;
  }
  return contextMessage;
}


/* ── B5. Thinking message detector ──────────────────────────────────
   Detects short "thinking" messages and returns an instant local
   reply without calling the Worker — saving an API call.
────────────────────────────────────────────────────────────────────── */

const THINKING_MESSAGES = new Set([
  "hmm", "hm", "hmm...", "hm...", "hmmmm",
  "let me think", "one sec", "one moment", "hold on",
  "wait", "ok", "okay", "k", "...", ". . .",
]);

const THINKING_REPLY = "Take your time — I'm right here whenever you're ready.";

/**
 * Returns an instant reply for thinking messages, or null if not a
 * thinking message (caller should proceed to FAQ/AI layer normally).
 *
 * @param {string} message
 * @returns {string | null}
 */
export function getThinkingReply(message) {
  const lower = message.toLowerCase().trim();
  if (THINKING_MESSAGES.has(lower)) return THINKING_REPLY;
  // Also catch very short unclear messages (1-3 chars that aren't keywords)
  if (lower.length <= 3 && /^[.!?…]+$/.test(lower)) return THINKING_REPLY;
  return null;
}


/* ── B6. Idle nudge ──────────────────────────────────────────────────
   After 45 seconds of inactivity (chat open, no message sent),
   the bot sends one gentle nudge. Never repeats.
────────────────────────────────────────────────────────────────────── */

const IDLE_DELAY_MS  = 45_000;
const IDLE_MESSAGES  = [
  "Still there? Happy to answer anything about Mossaic or bookMySlot 😊",
  "No rush — whenever you're ready, I'm here for any Mossaic questions.",
  "Take your time. If you'd like, I can walk you through bookMySlot's key features.",
];

let _idleTimer   = null;
let _idleFired   = false;

/**
 * Start the idle nudge timer.
 * Call this when the chat window opens.
 *
 * @param {Function} onNudge — callback that receives the nudge message string
 */
export function startIdleTimer(onNudge) {
  stopIdleTimer();
  _idleFired = false;
  _idleTimer = setTimeout(() => {
    if (!_idleFired) {
      _idleFired = true;
      const msg = IDLE_MESSAGES[Math.floor(Math.random() * IDLE_MESSAGES.length)];
      onNudge(msg);
    }
  }, IDLE_DELAY_MS);
}

/** Stop + reset the idle timer (call on every user message). */
export function stopIdleTimer() {
  if (_idleTimer) { clearTimeout(_idleTimer); _idleTimer = null; }
}

/** Reset idle state (call on chat close/reset). */
export function resetIdleTimer() {
  stopIdleTimer();
  _idleFired = false;
}


/* ── B7. Master send function ────────────────────────────────────────
   This replaces your current handleUserSend() / askMossie() call.
   Wires all the additions above into one clean pipeline.

   Call this from your chat UI's send handler, passing:
   • userMessage  — raw string from the input field
   • matchFaq     — your existing FAQ matcher function (returns string|null)
   • onBotReply   — callback that receives the final reply string
   • onTyping     — callback(true/false) to show/hide typing indicator
────────────────────────────────────────────────────────────────────── */

export async function handleSend(userMessage, matchFaq, onBotReply, onTyping) {
  const raw = userMessage.trim();
  if (!raw) return;

  // Reset idle timer on every send
  stopIdleTimer();

  // B5: Thinking message? Reply instantly, no API call.
  const thinkingReply = getThinkingReply(raw);
  if (thinkingReply) {
    await delayMs(400); // tiny pause so it doesn't feel instant-robotic
    onBotReply(thinkingReply);
    return;
  }

  // B2: Capture name if present
  captureUserName(raw);

  // Layer 1: FAQ matcher (your existing client-side logic)
  const faqAnswer = matchFaq(raw);
  if (faqAnswer) {
    onBotReply(faqAnswer);
    return;
  }

  // Layer 2: AI fallback — build enriched message for Worker
  let enriched = injectContextWithName(raw);   // B1 + B2
  enriched      = injectEmpathyIfNeeded(enriched, raw);  // B3
  enriched      = injectLanguageIfNeeded(enriched, raw); // B4

  // Show typing indicator
  onTyping(true);

  try {
    const reply = await askMossie(enriched); // your existing Worker call
    onTyping(false);
    onBotReply(reply);
  } catch {
    onTyping(false);
    onBotReply("I'm having a moment — please try again or email connect@mossaic.in.");
  }
}

function delayMs(ms) {
  return new Promise((r) => setTimeout(r, ms));
}

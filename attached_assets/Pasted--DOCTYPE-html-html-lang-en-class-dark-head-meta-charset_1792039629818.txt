    let currentSlide = 0;

    // Slide navigation

      currentSlide = (currentSlide + 1) % totalSlides;

      currentSlide = (currentSlide - 1 + totalSlides) % totalSlides;
      updateSlideDisplay();
    }

    function updateSlideDisplay() {
      document.getElementById('currentSlideNum').innerText = currentSlide + 1;

      for (let i = 0; i < totalSlides; i++) {
        const slide = document.getElementById(`slide-${i}`);
        const tab = document.getElementById(`tab-${i}`);

        if (i === currentSlide) {
          slide.classList.remove('hidden', 'opacity-0', 'scale-95');
          slide.classList.add('opacity-100', 'scale-100');
          
          tab.classList.remove('bg-slate-800', 'text-slate-400');
          tab.classList.add('bg-sky-500', 'text-slate-950', 'shadow-lg');
        } else {
          slide.classList.add('hidden', 'opacity-0', 'scale-95');
          slide.classList.remove('opacity-100', 'scale-100');

          tab.classList.remove('bg-sky-500', 'text-slate-950', 'shadow-lg');
          tab.classList.add('bg-slate-800', 'text-slate-400');
        }
      }
    }

      const toggleIcon = document.getElementById('toggleIcon');

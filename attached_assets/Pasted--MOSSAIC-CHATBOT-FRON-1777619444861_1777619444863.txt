/* ════════════════════════════════════════════════════════════════════
   1.  CONVERSATION HISTORY STORE
   ════════════════════════════════════════════════════════════════════
   Keep this in module scope (or React state / a ref) so it persists
   across calls within the same chat session.
   ════════════════════════════════════════════════════════════════════ */

/** @type {{ role: "user" | "assistant", content: string }[]} */
let conversationHistory = [];

const MAX_HISTORY_KEPT = 12; // 6 user + 6 assistant turns stored client-side

/**
 * Append a turn to the local history buffer.
 * Trims to MAX_HISTORY_KEPT so we don't send a huge payload on long sessions.
 */
function pushHistory(role, content) {
  conversationHistory.push({ role, content });
  if (conversationHistory.length > MAX_HISTORY_KEPT) {
    conversationHistory = conversationHistory.slice(-MAX_HISTORY_KEPT);
  }
}

/** Call this when the chat window is closed or reset. */
function clearHistory() {
  conversationHistory = [];
}


/* ════════════════════════════════════════════════════════════════════
   2.  CLIENT-SIDE PRE-GUARDS
   ════════════════════════════════════════════════════════════════════
   Run these BEFORE calling the Worker — saves an API call for
   clearly off-topic messages that slip past your FAQ matcher.
   These mirror Layer 3 & 4 in the Worker but are faster (local).
   ════════════════════════════════════════════════════════════════════ */

const CLIENT_BOUNDARY_MSG =
  "I'm only set up to help with Mossaic and bookMySlot questions. " +
  "Anything else, the team is at connect@mossaic.in.";

const CLIENT_JAILBREAK_MSG =
  "I'm a focused FAQ helper for Mossaic — I can't step outside that role. " +
  "Is there something about our products I can help with?";

/** Fast client-side off-topic check (mirrors Worker Layer 3). */
const CLIENT_OFFTOPIC = [
  /\b(chatgpt|openai|gpt-?4|gemini|copilot|claude|mistral)\b/i,
  /\b(act as|pretend (you are|to be)|roleplay|forget (your|the) (instructions?|rules?))\b/i,
  /\b(ignore (previous|all|your) (instructions?|rules?|prompt))\b/i,
  /\b(jailbreak|dan mode|developer mode|unrestricted mode)\b/i,
  /\b(diagnose (me|my)|what (disease|illness) do i|am i sick)\b/i,
  /\b(should i invest|stock tips?|legal advice|sue|lawsuit)\b/i,
  /\b(porn|sex|nude|violence|kill|murder|hack)\b/i,
];

const CLIENT_JAILBREAK = [
  /system prompt/i,
  /your (instructions|rules|guidelines|constraints)/i,
  /repeat after me/i,
  /\bignore\b.*\brules?\b/i,
];

function clientIsOffTopic(text) {
  return CLIENT_OFFTOPIC.some((re) => re.test(text));
}

function clientIsJailbreak(text) {
  return CLIENT_JAILBREAK.some((re) => re.test(text));
}


/* ════════════════════════════════════════════════════════════════════
   3.  WORKER CALL — MAIN FUNCTION
   ════════════════════════════════════════════════════════════════════
   Replace your existing fetch() call with this.
   ════════════════════════════════════════════════════════════════════ */

const WORKER_URL =
  typeof import !== "undefined" && typeof import.meta !== "undefined"
    ? import.meta.env?.VITE_WORKER_URL
    : "https://your-worker.your-subdomain.workers.dev"; // ← hardcode for static sites

const WORKER_TIMEOUT_MS = 8000; // abort if Groq takes longer than 8s

/**
 * Send a message to the Mossaic FAQ Worker.
 *
 * @param {string} userMessage  — the raw text the user typed
 * @returns {Promise<string>}   — the bot's reply
 */
async function askMossie(userMessage) {
  const msg = userMessage.trim();

  /* 2a. Empty guard */
  if (!msg) return "It looks like your message was empty — what would you like to know?";

  /* 2b. Length guard */
  if (msg.length > 400) {
    return (
      "That's a bit long for me — could you summarise in a sentence or two? " +
      "I'm best with specific Mossaic questions."
    );
  }

  /* 2c. Client-side off-topic / jailbreak guards */
  if (clientIsOffTopic(msg)) return CLIENT_BOUNDARY_MSG;
  if (clientIsJailbreak(msg)) return CLIENT_JAILBREAK_MSG;

  /* 2d. Add user turn to history */
  pushHistory("user", msg);

  /* 2e. Call the Worker */
  try {
    const controller = new AbortController();
    const timeoutId  = setTimeout(() => controller.abort(), WORKER_TIMEOUT_MS);

    const response = await fetch(WORKER_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      signal: controller.signal,
      body: JSON.stringify({
        message: msg,
        history: conversationHistory.slice(0, -1), // exclude the turn we just added
      }),
    });

    clearTimeout(timeoutId);

    if (!response.ok) {
      console.error("Worker HTTP error:", response.status);
      return "I'm having a moment — please try again, or email connect@mossaic.in.";
    }

    const data = await response.json();
    const reply = data?.reply;

    if (!reply || typeof reply !== "string" || reply.trim().length === 0) {
      return "I couldn't form a good answer just then — please try rephrasing, or email connect@mossaic.in.";
    }

    /* 2f. Add assistant turn to history */
    pushHistory("assistant", reply.trim());

    return reply.trim();

  } catch (err) {
    if (err.name === "AbortError") {
      return "I'm taking too long to respond — please try again or email connect@mossaic.in.";
    }
    console.error("Worker fetch error:", err);
    return "Something went wrong on my end — please try again, or email connect@mossaic.in.";
  }
}


/* ════════════════════════════════════════════════════════════════════
   4.  TYPING ANIMATION + WORD REVEAL
   ════════════════════════════════════════════════════════════════════
   Call showTypingThenReveal() instead of directly setting the message
   text. Works with any DOM element or React state setter.
   ════════════════════════════════════════════════════════════════════ */

const TYPING_DELAY_MS  = 900;   // how long the "..." bubble shows
const WORD_INTERVAL_MS = 28;    // ms between each word appearing

/**
 * Show a typing indicator for TYPING_DELAY_MS, then reveal the reply
 * word-by-word so it feels streamed.
 *
 * @param {string}   reply        — the final reply string
 * @param {Function} onTyping     — called with true to show dots, false to hide
 * @param {Function} onWordUpdate — called with the growing partial string
 * @param {Function} onDone       — called with the final string when complete
 */
async function showTypingThenReveal(reply, onTyping, onWordUpdate, onDone) {
  onTyping(true);
  await delay(TYPING_DELAY_MS);
  onTyping(false);

  const words = reply.split(" ");
  let partial  = "";

  for (let i = 0; i < words.length; i++) {
    partial += (i === 0 ? "" : " ") + words[i];
    onWordUpdate(partial);
    await delay(WORD_INTERVAL_MS);
  }

  onDone(reply);
}

function delay(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}


/* ════════════════════════════════════════════════════════════════════
   5.  REACT USAGE EXAMPLE
   ════════════════════════════════════════════════════════════════════
   Shows how to wire askMossie + showTypingThenReveal into your
   existing Chatbot component. Adapt to your own state shape.
   ════════════════════════════════════════════════════════════════════

import { useState, useRef } from "react";

export function useMossieAI() {
  const [isTyping, setIsTyping]   = useState(false);
  const [aiReply,  setAiReply]    = useState("");
  const isBusyRef = useRef(false);

  async function sendToMossie(userMessage, faqMatchedAlready) {
    // If your FAQ matcher already found a perfect match, skip the Worker.
    // The Worker is only for questions that slip past the FAQ layer.
    if (faqMatchedAlready) return;

    if (isBusyRef.current) return; // prevent double-sends
    isBusyRef.current = true;

    setIsTyping(true);
    setAiReply("");

    try {
      const reply = await askMossie(userMessage);

      await showTypingThenReveal(
        reply,
        (typing) => setIsTyping(typing),
        (partial) => setAiReply(partial),
        (final)   => {
          setAiReply(final);
          isBusyRef.current = false;
        }
      );
    } catch {
      setIsTyping(false);
      setAiReply("Something went wrong — please email connect@mossaic.in.");
      isBusyRef.current = false;
    }
  }

  return { isTyping, aiReply, sendToMossie, clearHistory };
}

════════════════════════════════════════════════════════════════════ */

async function handleUserSend(userMessage) {
  const faqAnswer = matchFaq(userMessage); // your existing FAQ matcher

  await showTypingThenReveal(

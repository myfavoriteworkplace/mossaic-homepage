3. Data Schema & Runtime Validation (Zod)
To ensure that manual edits to JSON data files never cause runtime render crashes or silent layout bugs, validate all imported data through Zod schemas.
A. Products Schema (src/types/product.ts)
TypeScript
import { z } from 'zod';

export const ProductSchema = z.object({
  id: z.string(),
  title: z.string(),
  category: z.string(),
  badge: z.string(),
  status: z.enum(['Live', 'Coming soon', '2027', 'Available']),
  description: z.string(),
  tags: z.array(z.string()),
  iconName: z.string(), // Key mapping directly to Lucide React icons
});

export type Product = z.infer<typeof ProductSchema>;
B. Testimonials Schema (src/types/testimonial.ts)
TypeScript
import { z } from 'zod';

export const TestimonialSchema = z.object({
  id: z.string(),
  productTag: z.string(),
  rating: z.number().min(1).max(5),
  quote: z.string(),
  author: z.object({
    name: z.string(),
    initials: z.string(),
    clinicName: z.string(),
    city: z.string(),
  }),
  isFeatured: z.boolean(),
});

export type Testimonial = z.infer<typeof TestimonialSchema>;

A. Dynamic Products Grid (src/components/ProductsGrid.tsx)

import React from 'react';
import { motion } from 'framer-motion';
import * as LucideIcons from 'lucide-react';
import rawProductsData from '../data/products.json';
import { ProductSchema, Product } from '../types/product';

// Validate JSON on module load
const products: Product[] = rawProductsData.map((p) => ProductSchema.parse(p));

export const ProductsGrid: React.FC = () => {
  return (
    <section className="py-20 px-6 max-w-7xl mx-auto space-y-12">
      <div className="text-center space-y-3">
        <span className="text-xs font-bold text-emerald-400 border border-emerald-500/20 bg-emerald-500/10 px-3 py-1 rounded-full uppercase tracking-wider">
          OUR PRODUCTS & SERVICES
        </span>
        <h2 className="text-3xl md:text-5xl font-extrabold text-white">
          One company. <span className="text-emerald-400">Products & systems that work.</span>
        </h2>
        <p className="text-slate-400 text-sm max-w-xl mx-auto">
          Deep vertical software alongside custom-engineered enterprise platforms.
        </p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6">
        {products.map((item, index) => {
          // Dynamic Lucide Icon Resolution
          const IconComponent = (LucideIcons as Record<string, React.ElementType>)[item.iconName] || LucideIcons.Box;

          return (
            <motion.div
              key={item.id}
              initial={{ opacity: 0, y: 20 }}
              whileInView={{ opacity: 1, y: 0 }}
              transition={{ duration: 0.4, delay: index * 0.1 }}
              viewport={{ once: true }}
              whileHover={{ y: -4 }}
              className="bg-slate-900/60 border border-slate-800 hover:border-emerald-500/40 rounded-2xl p-6 flex flex-col justify-between transition-all group backdrop-blur-md shadow-xl"
            >
              <div>
                <div className="flex items-center justify-between mb-4">
                  <div className="w-12 h-12 rounded-xl bg-emerald-500/10 border border-emerald-500/20 text-emerald-400 flex items-center justify-center">
                    <IconComponent className="w-6 h-6" />
                  </div>
                  <span className="bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 text-xs font-semibold px-2.5 py-1 rounded-full">
                    {item.badge}
                  </span>
                </div>

                <h3 className="text-lg font-bold text-white group-hover:text-emerald-400 transition-colors">
                  {item.title}
                </h3>
                <div className="text-xs font-medium text-slate-400 mt-0.5 mb-3">

                  {item.description}

B. Dynamic Testimonials Section (src/components/TestimonialsSection.tsx)
TypeScript
import React from 'react';
import { motion } from 'framer-motion';
import { Star } from 'lucide-react';
import rawTestimonialsData from '../data/testimonials.json';
import { TestimonialSchema, Testimonial } from '../types/testimonial';

const testimonials: Testimonial[] = rawTestimonialsData.map((t) => TestimonialSchema.parse(t));

export const TestimonialsSection: React.FC = () => {
  return (
    <section className="py-20 px-6 max-w-7xl mx-auto space-y-12">
      <div className="text-center space-y-3">
        <span className="text-xs font-bold text-emerald-400 uppercase tracking-widest bg-emerald-500/10 border border-emerald-500/20 px-3 py-1 rounded-full">
          FROM OUR PRODUCTS IN MARKET
        </span>
        <h2 className="text-3xl md:text-4xl font-extrabold text-white">
          What customers are <span className="text-emerald-400">saying.</span>
        </h2>
        <p className="text-slate-400 text-sm max-w-xl mx-auto">
          Quotes from clinics using bookMySlot and enterprises built with mossaic Studio.
        </p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
        {testimonials.filter(t => t.isFeatured).map((item, index) => (
          <motion.div
            key={item.id}
            initial={{ opacity: 0, scale: 0.95 }}
            whileInView={{ opacity: 1, scale: 1 }}
            transition={{ duration: 0.4, delay: index * 0.1 }}
            viewport={{ once: true }}
            className="bg-slate-900/60 border border-slate-800 hover:border-slate-700 rounded-2xl p-6 flex flex-col justify-between space-y-6 shadow-lg backdrop-blur-md"
          >
            <div className="space-y-4">
              <div className="flex justify-between items-center">
                <div className="flex text-amber-400 gap-1">
                  {Array.from({ length: item.rating }).map((_, i) => (
                    <Star key={i} className="w-4 h-4 fill-amber-400" />
                  ))}
                </div>
                <span className="text-[10px] font-mono bg-emerald-500/10 border border-emerald-500/20 text-emerald-400 px-2.5 py-1 rounded-md font-semibold">
                  {item.productTag}
                </span>
              </div>
              <p className="text-xs text-slate-300 leading-relaxed italic">
                "{item.quote}"
              </p>
            </div>

            <div className="flex items-center gap-3 pt-4 border-t border-slate-800/80">
              <div className="w-10 h-10 rounded-full bg-emerald-500/20 border border-emerald-500/30 text-emerald-400 font-bold flex items-center justify-center text-xs">
                {item.author.initials}
              </div>
              <div>
                <div className="text-xs font-bold text-white">{item.author.name}</div>
                <div className="text-[11px] text-slate-400">
                  {item.author.clinicName}, {item.author.city}
                </div>
              </div>
            </div>
          </motion.div>
        ))}
      </div>
    </section>
  );
};
